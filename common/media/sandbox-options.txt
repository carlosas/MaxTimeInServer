VERSION = 1,

/*
 * Server settings for MaxTimeInServer. They appear under their own page in the
 * server's sandbox options, so they can be set from the admin panel or by
 * editing <World>_SandboxVars.lua, and are read when the server boots: changing
 * one takes a restart. They show as these raw keys because the file that would
 * give them readable names belongs to the base game, and a mod shipping it
 * shadows hundreds of the game's own entries.
 */

option MaxTimeInServer.CapMinutes
{
    type = integer,
    min = 1,
    max = 100000,
    default = 600,
    page = MaxTimeInServer,
    translation = MaxTimeInServer_CapMinutes,
}

option MaxTimeInServer.ExemptRoles
{
    type = string,
    default = admin;moderator;gm,
    page = MaxTimeInServer,
    translation = MaxTimeInServer_ExemptRoles,
}

/*
 * Left empty, the mod says it in each player's own language. Set it to use your
 * own wording instead, which every player then reads exactly as written. It
 * cannot contain a comma: the parser splits a block on them. Either way the
 * numbers - minutes used, the cap, when it resets - follow underneath,
 * translated.
 */
option MaxTimeInServer.BlockedMessage
{
    type = string,
    default = ,
    page = MaxTimeInServer,
    translation = MaxTimeInServer_BlockedMessage,
}

option MaxTimeInServer.GraceSeconds
{
    type = integer,
    min = 0,
    max = 300,
    default = 12,
    page = MaxTimeInServer,
    translation = MaxTimeInServer_GraceSeconds,
}
