VERSION = 1,

/*
 * Server settings for MaxTimeInServer. They appear under their own page in the
 * server's sandbox options, so they can be set from the admin panel or by
 * editing <World>_SandboxVars.lua. Sandbox options are read when the server
 * boots, so changing one takes a restart.
 *
 * The option names show as these raw keys rather than as prose. Giving them
 * readable names needs a media/lua/shared/Translate/<LANG>/Sandbox.json, and
 * that path belongs to the base game: shipping one from a mod shadows the
 * game's own file, which holds hundreds of keys. Raw keys are the lesser
 * problem.
 */

option MaxTimeInServer.CapMinutes
{
    type = integer,
    min = 1,
    max = 100000,
    default = 600,
    page = MaxTimeInServer,
    translation = MaxTimeInServer_CapMinutes,
}

option MaxTimeInServer.ExemptRoles
{
    type = string,
    default = admin;moderator;gm,
    page = MaxTimeInServer,
    translation = MaxTimeInServer_ExemptRoles,
}

/*
 * The parser splits a block on commas, so this value cannot contain one. The
 * numbers - minutes used, the cap, when it resets - are added by the mod on
 * their own lines underneath, so this is only the explanation.
 */
option MaxTimeInServer.BlockedMessage
{
    type = string,
    default = Weekly play limit reached.,
    page = MaxTimeInServer,
    translation = MaxTimeInServer_BlockedMessage,
}

option MaxTimeInServer.GraceSeconds
{
    type = integer,
    min = 0,
    max = 300,
    default = 12,
    page = MaxTimeInServer,
    translation = MaxTimeInServer_GraceSeconds,
}
